\chapter*{Định Nghĩa và Viết Tắt}
\addcontentsline{toc}{chapter}{Định Nghĩa và Viết Tắt}

\begin{longtable}{|>{\bfseries\raggedright\arraybackslash}p{1.5cm}|>{\raggedright\arraybackslash}p{5.0cm}|>{\raggedright\arraybackslash}p{7.5cm}|}
  \hline
  \textbf{Viết tắt} & \textbf{Định nghĩa} & \textbf{Mô tả} \\
  \hline
  \endfirsthead
  \hline
  \textbf{Viết tắt} & \textbf{Định nghĩa} & \textbf{Mô tả} \\
  \hline
  \endhead
  \hline
  \endfoot
  \hline
  \endlastfoot
  AI    & Artificial Intelligence &
         Trí tuệ nhân tạo, công nghệ mô phỏng các quá trình suy nghĩ và học tập của con người. \\ \hline
  API   & Application Programming Interface &
         Tập hợp quy tắc cho phép các ứng dụng giao tiếp với nhau. \\ \hline
  BA    & Business Analysis &
         Xác định nhu cầu và đề xuất giải pháp mang lại giá trị cho các bên liên quan. \\ \hline
  BR    & Business Rule &
         Quy tắc xác định hoặc ràng buộc một khía cạnh của nghiệp vụ. \\ \hline
  BVLOS & Beyond Visual Line of Sight &
         Bay ngoài tầm nhìn trực quan, phương thức vận hành drone không yêu cầu người điều khiển nhìn thấy thiết bị. \\ \hline
  CCU   & Concurrent Users &
         Số lượng người dùng truy cập và tương tác với hệ thống đồng thời tại cùng một thời điểm. \\ \hline
  CSDL  & Cơ sở dữ liệu &
         Tập hợp dữ liệu có cấu trúc được lưu trữ và truy cập thông qua hệ quản trị cơ sở dữ liệu. \\ \hline

  ERD   & Entity Relationship Diagram &
         Sơ đồ mô tả mối quan hệ giữa các thực thể trong hệ thống. \\ \hline
  ETA   & Estimated Time of Arrival &
         Thời gian dự kiến đến, chỉ thời gian drone dự kiến hoàn thành hành trình giao hàng. \\ \hline
  FR    & Functional Requirement &
         Yêu cầu chức năng, mô tả các tính năng mà hệ thống phần mềm phải thực hiện. \\ \hline
  GPS   & Global Positioning System &
         Hệ thống định vị toàn cầu, dùng để xác định tọa độ địa lý của drone và trạm hạ cánh. \\ \hline

  HTTPS & Hypertext Transfer Protocol Secure &
         Giao thức truyền tải siêu văn bản an toàn, mã hóa dữ liệu truyền tải giữa trình duyệt và máy chủ. \\ \hline
  JWT   & JSON Web Token &
         Mã thông báo dạng chuỗi JSON dùng để xác thực và chia sẻ thông tin an toàn giữa các bên. \\ \hline
  LLM   & Large Language Model &
         Mô hình ngôn ngữ lớn, mô hình AI được huấn luyện trên lượng dữ liệu lớn để xử lý ngôn ngữ tự nhiên. \\ \hline
  OWASP & Open Web Application Security Project &
         Dự án bảo mật ứng dụng web mở, tổ chức phi lợi nhuận cung cấp các tiêu chuẩn bảo mật phần mềm. \\ \hline

  RAM   & Random Access Memory &
         Bộ nhớ truy cập ngẫu nhiên của thiết bị phần cứng. \\ \hline
  SDD   & Software Design Description &
         Tài liệu mô tả kiến trúc và thiết kế chi tiết hệ thống phần mềm. \\ \hline

  SQL   & Structured Query Language &
         Ngôn ngữ truy vấn có cấu trúc dùng để quản lý dữ liệu trong hệ quản trị cơ sở dữ liệu quan hệ. \\ \hline
  SRS   & Software Requirement Specification &
         Tài liệu đặc tả yêu cầu phần mềm, mô tả hệ thống cần làm gì. \\ \hline
  TC    & Test Case &
         Ca kiểm thử, tập hợp các điều kiện và các bước thực hiện để xác minh tính đúng đắn của phần mềm. \\ \hline
  UAT   & User Acceptance Test &
         Kiểm thử chấp nhận người dùng -- giai đoạn kiểm thử cuối cùng. \\ \hline
  UC    & Use Case &
         Trường hợp sử dụng, mô tả tương tác giữa người dùng và hệ thống. \\ \hline
  UUID  & Universally Unique Identifier &
         Mã định danh duy nhất toàn cầu, chuỗi ký tự ngẫu nhiên dùng để định danh tài nguyên hệ thống. \\ \hline
  XSS   & Cross-Site Scripting &
         Tấn công chèn kịch bản chéo trang, một trong những lỗ hổng bảo mật web phổ biến. \\ \hline
\end{longtable}
